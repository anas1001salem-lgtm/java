package edu.zuj.oop.week2;

import java.util.Scanner;

public class DiamondOfStars {
    public static void main(String[] args) {
    Scanner s=new Scanner(System.in);
        System.out.print("Enter a number: ");
        int stars = 1,space = 0,n=s.nextInt(),loop=0;
        
        for(int i=0;i<n;i++){
            for(int q=0;space<n-1;space++){
                System.out.print(" ");                
            }
            for(int j=0;j<stars;j++){
                System.out.print("*");                
            }
            System.out.println();
            stars+=2;
            space=1+i;                        
        }
        stars=(n*2)-1;
        space=n;
        for(int i=0;i<n-1;i++){
            loop=0;
            stars-=2;
            for(int q=0;space<=n;space++){
                System.out.print(" "); 
                loop++;
            }
            for(int j=0;j<stars;j++){
                System.out.print("*");                
            }                        
            System.out.println();            
            space -= (loop+1);                                   
        }
    }
}
