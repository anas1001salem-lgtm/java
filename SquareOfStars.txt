package edu.zuj.oop.week2;

import java.util.Scanner;

public class SquareOfStars {
    public static void main(String[] args) {
        Scanner s=new Scanner(System.in);
        System.out.print("Enter number: ");
        int n=s.nextInt();
        for(int i=0; i<n ;i++){
            for(int j=1; j<n;j++){
                System.out.print("*");
            }
            System.out.println("*");
        }
            
    }
}
