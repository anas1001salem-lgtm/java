package edu.zuj.oop.week2;

import java.util.Scanner;

public class NumberStats {
    public static void main(String[] args) {
        Scanner s=new Scanner(System.in);
        System.out.print("Enter number of items: ");
        
        int n=s.nextInt();
        System.out.println("=============================");
        int a[]=new int[n];
        int max=0;
        int min=0;
        double avg=0;
        int sum=0;
        for(int i=0;i<n;i++){
            System.out.print("Enter the value of item "+(i+1)+": ");
            a[i]=s.nextInt();
        }
        System.out.println("=============================");
        for(int i=0;i<n;i++){         
            if(i==0)
                max=a[i];
            else if (a[i]>a[i-1])
                max=a[i];
            else
                break;
        }
        for(int i=0;i<n;i++){
            if(i==0)
                min=a[i];
            else if (a[i]<a[i-1])
                min=a[i];
            else
                break;
        }
        for(int i=0;i<n;i++){
            sum+=a[i];
        }
        for(int i=0;i<n;i++){
            avg= (double)sum/(double)n;
        }                
        System.out.println("The max number = "+max);
        System.out.println("The min number = "+min);
        System.out.println("The sum = "+sum);
        System.out.println("The avg = "+avg);
        for(int i=0;i<n;i++){
            System.out.println(a[i]);
        }
    }
}
