package edu.zuj.oop.week2;

import java.util.Scanner;

public class HorizontalLineOfStars {
    public static void main(String[] args) {
        Scanner s=new Scanner(System.in);
        System.out.print("Enter number: ");
        int n=s.nextInt();
        for(int i=0;i<n;i++){
            System.out.print("*");
        }
    }
}
