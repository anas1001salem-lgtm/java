package edu.zuj.oop.week2;

import java.util.Scanner;

public class OddNumbers {
    public static void main(String[] args) {
        Scanner s=new Scanner(System.in);
        System.out.print("Enter Number: ");
        int n=s.nextInt();
        for(int i=0;i<=n;i++){
            if(i%2!=0){
                System.out.println(i+" Is odd");
            }
        
    }
    }
}
